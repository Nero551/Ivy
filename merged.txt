struct Camera : C::Component
{
    float FOV = 45.0f;
    float Near = -1;
    float Far = 1;
    float Left = -50;
    float Right = 50;
    float Top = 50;
    float Bottom = -50;
    R::ProjectionMode ProjectionMode = R::ProjectionMode::Orthographic;
    float Sensitivity = 0.05f;

    M::Matrix<4, 4> GetProjectionMatrix() const
    {
        if (ProjectionMode == R::ProjectionMode::Orthographic)
        {
            return M::Matrix<4, 4>::Orthographic(Left, Right, Bottom, Top, Near, Far);
        }
        return M::Matrix<4, 4>::Perspective(M::Rad(FOV), (Right - Left) / (Top - Bottom), Near, Far);
    }
};

struct Light : C::Component
{
    R::LightType Type = R::LightType::Directional;
    M::Vector<3> Color = M::Vector<3>{1};
    M::Vector<3> Ambient = M::Vector<3>{0.8};
    M::Vector<3> Diffuse = M::Vector<3>{1};
    M::Vector<3> Specular = M::Vector<3>{1};

    float Intensity = 1;
    float Constant = 1.0f;
    float Linear = 0.009f;
    float Quadratic = 0.032f;
    float InnerCutOff = M::Rad(7.5);
    float OuterCutOff = M::Rad(17.5);
};

};

namespace Ivy::R

struct InstanceData
{
    M::Matrix<4, 4> ModelMatrix;
    M::Matrix<3, 3> NormalMatrix;

    InstanceData(const M::Matrix<4, 4>& model, const M::Matrix<3, 3>& normal)
        : ModelMatrix(model), NormalMatrix(normal)
    {
    }
};

struct RenderBatch
{
    U::CheckedPtr<C::Resource<G::Material>> Material;
    U::CheckedPtr<C::Resource<G::Mesh>> Mesh;
    U::SparseSetSoA<InstanceData> Instances;
    G::ArrayBuffer Buffer;
    G::VertexArray VAO;

    RenderBatch(
        const U::CheckedPtr<C::Resource<G::Mesh>>& mesh, const U::CheckedPtr<C::Resource<G::Material>>& mat)
        : Material(mat), Mesh(mesh)
    {
        Buffer.Usage = G::BufferUsage::DynamicDraw;
        Buffer.Generate();
    }

    void Render()
    {
        Material->Get().Use();

        unsigned int instanceCount = Instances.Size();
        Buffer.SetData(Instances.Data());

        Mesh->Get().VAO.SetVertexBuffer(Buffer, 1, sizeof(InstanceData));

        Mesh->Get().VAO.SetMatrix4AttribPointer(4, offsetof(InstanceData, ModelMatrix), 1);
        Mesh->Get().VAO.SetMatrix3AttribPointer(8, offsetof(InstanceData, NormalMatrix), 1);
        Mesh->Get().VAO.SetAttribDivisor(1, 1);

        Mesh->Get().DrawInstanced(instanceCount);
    }

    using BatchKey = std::pair<struct G::Material*, struct G::Mesh*>;
    struct BatchKeyHash
    {
        std::size_t operator()(const BatchKey& key) const
        {
            const std::size_t h1 = std::hash<struct G::Material*>{}(key.first);
            const std::size_t h2 = std::hash<struct G::Mesh*>{}(key.second);

            return h1 ^ (h2 << 1U);
        }
    };
};
} // namespace Ivy::R

void Renderer::SetupFramebuffer()
{
    const std::vector<G::Vertex> vertices = {G::Vertex({-1.0f, -1.0f, 0.0f, 1.0f}, {}, {0.0f, 0.0f}, {}),
        G::Vertex({1.0f, -1.0f, 0.0f, 1.0f}, {}, {1.0f, 0.0f}, {}),
        G::Vertex({1.0f, 1.0f, 0.0f, 1.0f}, {}, {1.0f, 1.0f}, {}),
        G::Vertex({-1.0f, 1.0f, 0.0f, 1.0f}, {}, {0.0f, 1.0f}, {})};

    const std::vector<unsigned int> indices = {0, 1, 2, 2, 3, 0};
    auto& resources = m_Engine.GetService<C::ResourceManager>();
    auto& window = m_Engine.Window;

    auto& screenShader = resources.Load<G::Shader>("[Graphics] Screen Shader");
    auto& screenVert = resources.Load<G::ShaderSource>(
        "[Graphics] Screen Vertex", "Assets/Shaders/screen.vert", G::ShaderStage::Vertex);
    auto& screenFrag = resources.Load<G::ShaderSource>(
        "[Graphics] Screen Fragment", "Assets/Shaders/screen.frag", G::ShaderStage::Fragment);

    screenShader().AssignSource(screenVert);
    screenShader().AssignSource(screenFrag);

    ScreenMaterial.Shader = &screenShader();
    ScreenMaterial.Depth.Enabled = false;
    ScreenMaterial.Stencil.Enabled = false;
    ScreenMaterial.Blend.Enabled = false;

    ScreenMesh.Vertices = vertices;
    ScreenMesh.Indices = indices;
    ScreenMesh.CullMode = G::CullMode::None;

    auto& screenTexture = resources.Load<G::Texture2D>("COLOR_BUFFER");
    screenTexture.Get().Width = window.GetWidth();
    screenTexture.Get().Height = window.GetHeight();
    screenTexture.Get().InternalFormat = G::TextureInternalFormat::RGB8;
    screenTexture.Get().Format = G::TextureFormat::RGB;
    screenTexture.Get().DataType = G::DataType::UnsignedByte;
    screenTexture.Get().AutoMipmaps = false;
    screenTexture.Get().MagFilter = G::TextureFilter::Linear;
    screenTexture.Get().MinFilter = G::TextureFilter::Linear;

    Framebuffer.AttachTexture(G::FramebufferAttachment::Color0, screenTexture);

    m_Engine.Window.SetFramebufferSizeCallback(
        [&](G::Window& win, const int w, const int h)
        {
            G::Context::SetViewport(0, 0, w, h);
            auto& graphics = m_Engine.GetModule<Renderer>();
            graphics.MSAntiAliasing.OutputBuffer.Resize(w, h);
            graphics.Framebuffer.Resize(w, h);
        });
}

void Renderer::PresentFramebuffer()
{
    auto& window = m_Engine.Window;
    int width = window.GetWidth();
    int height = window.GetHeight();

    MSAntiAliasing.OutputBuffer.Blit(Framebuffer, width, height, width, height, G::BufferBit::Color);

    MSAntiAliasing.OutputBuffer.Unbind();

    G::Context::ClearColor(0.08, 0.05, 0.1, 1);
    G::Context::Clear(G::BufferBit::Color);

    int i = 0;
    for (auto& texture : Framebuffer.TextureAttachments | std::views::values)
    {
        ScreenMaterial.AssignTexture(*texture, i);
        ++i;
    }

    ScreenMaterial.Use();
    ScreenMesh.Draw();
}

void Renderer::Start()
{
    AddSystem<LightingSystem>();
    AddSystem<Renderer3DSystem>();
    AddSystem<Renderer2DSystem>();

    G::Context::Enable(G::Capability::ProgramPointSize);
    G::Context::Enable(G::Capability::Multisample);

    GUniformbuffer.Size = 160;

    MSAntiAliasing.Generate(m_Engine.Window.GetWidth(), m_Engine.Window.GetHeight());
    SetupFramebuffer();

    GetSystem<LightingSystem>().Start();
    GetSystem<Renderer3DSystem>().Start();
    GetSystem<Renderer2DSystem>().Start();
}

void Renderer::BeginFrame(double dt)
{
    MSAntiAliasing.OutputBuffer.Bind();
    G::Context::ClearColor(0.08, 0.05, 0.1, 1);
    G::Context::Clear(G::BufferBit::Color | G::BufferBit::Depth | G::BufferBit::Stencil);
}

void Renderer::Stop()
{
    const G::Texture2D& texture =
        static_cast<G::Texture2D&>(Framebuffer.GetTexture(G::FramebufferAttachment::Color0));

    U::Image image = {texture.Width, texture.Height, U::Image::ColorChannels::RGB, texture.ReadPixels()};
    image.SaveToDiskPNG("Assets/LastFrame.png", true);
}

#include "LightingSystem.hpp"

using C::ResourceManager;

void LightingSystem::Start()
{
    auto& resources = m_Engine.GetService<ResourceManager>();

    LightingBuffer = &resources.Load<G::Uniformbuffer>("[Graphics] Lighting Buffer")();
    LightingBuffer->Binding = 1;

    constexpr size_t LightStride = 144;
    constexpr size_t HeaderSize = 16;

    LightingBuffer->Size = HeaderSize + m_Engine.World.MaxLights * LightStride;
}

void LightingSystem::Render()
{
    auto& world = m_Engine.World;

    int i = 0;
    constexpr unsigned int LightStride = 144;
    m_Engine.ECS.Query.ForEach<CMP::Light, CMP::Transform3D>(
        [&](unsigned int entityId, CMP::Light& lightComponent, CMP::Transform3D& transform)
        {
            const unsigned int LightOffset = 16 + i * LightStride;

            LightingBuffer->Set(static_cast<int>(lightComponent.Type), LightOffset + 0);
            LightingBuffer->Set(transform.GetForward(), LightOffset + 16, 4);
            LightingBuffer->Set(lightComponent.Color, LightOffset + 32, 4);
            LightingBuffer->Set(transform.Position, LightOffset + 48, 4);
            LightingBuffer->Set(lightComponent.Ambient, LightOffset + 64, 4);
            LightingBuffer->Set(lightComponent.Diffuse, LightOffset + 80, 4);
            LightingBuffer->Set(lightComponent.Specular, LightOffset + 96, 4);

            LightingBuffer->Set(lightComponent.Intensity, LightOffset + 112);
            LightingBuffer->Set(lightComponent.Constant, LightOffset + 116);
            LightingBuffer->Set(lightComponent.Linear, LightOffset + 120);
            LightingBuffer->Set(lightComponent.Quadratic, LightOffset + 124);
            LightingBuffer->Set(lightComponent.InnerCutOff, LightOffset + 128);
            LightingBuffer->Set(lightComponent.OuterCutOff, LightOffset + 132);

            ++i;
        });
    LightingBuffer->Set(i, 0);
    LightingBuffer->Bind();
}
} // namespace Ivy::R

void Renderer2DSystem::Start()
{
    m_Engine.ECS.EntityDestroyed.Sub(
        [this](const C::EntityDestroyed& event)
        {
            for (auto& batch : Batches)
            {
                if (batch.Instances.Contains(event.Entity->GetId().Index))
                {
                    batch.Instances.Erase(event.Entity->GetId().Index);
                }
            }
        });
}

void Renderer2DSystem::Render()
{
    const auto& camera = m_Engine.World.GetCamera();
    auto& query = m_Engine.ECS.Query;
    auto& rendererModule = m_Engine.GetModule<Renderer>();

    if (!query.HasComponent<CMP::Transform2D>(camera))
    {
        return;
    }

    const M::Matrix<4, 4> projection = query.GetComponent<CMP::Camera>(camera).GetProjectionMatrix();

    //View Matrix
    auto& transform = m_Engine.ECS.Query.GetComponent<CMP::Transform2D>(camera);
    M::Matrix<4, 4> view = transform.GetModelMatrix().Inverse();
    auto& camPos = transform.m_GlobalPosition;

    rendererModule.GUniformbuffer.Set(view, 0);
    rendererModule.GUniformbuffer.Set(projection, 64);
    rendererModule.GUniformbuffer.Set(m_Engine.GetTime(), 128);
    rendererModule.GUniformbuffer.Set(M::Vector<3>{camPos().x, camPos().y, 0}, 144);
    rendererModule.GUniformbuffer.Bind();

    query.ForEach<CMP::Material, CMP::Transform2D, CMP::Mesh>(
        [&](unsigned int entityId, CMP::Material& materialComponent, CMP::Transform2D& transform,
            CMP::Mesh& meshComponent)
        {
            if (materialComponent.Material->Get().Shader->HotReload == true)
            {
                materialComponent.Material->Get().Shader->Delete();
            }

            FillBatches(entityId, transform, materialComponent, meshComponent);
        });
    for (auto& batch : Batches)
    {
        batch.Render();
    }
}

void Renderer2DSystem::FillBatches(unsigned int entityId, CMP::Transform2D& transformComponent,
    CMP::Material& materialComponent, CMP::Mesh& meshComponent)
{
    unsigned int materialId = materialComponent.Material->GetHandle().Index;
    unsigned int meshId = meshComponent.Mesh->GetHandle().Index;

    auto batch = Batches.Find(materialId, meshId);

    if (batch == Batches.end())
    {
        batch = Batches.Emplace(materialId, meshId, meshComponent.Mesh, materialComponent.Material);
    }

    if (!batch->Instances.Contains(entityId) || transformComponent.m_GlobalPosition.IsChanged() ||
        transformComponent.m_GlobalRotation.IsChanged() || transformComponent.m_GlobalScale.IsChanged())
    {
        batch->Instances.EmplaceOrReplace(
            entityId, transformComponent.GetModelMatrix(), transformComponent.GetNormalMatrix());
        transformComponent.m_GlobalPosition.ClearChanged();
        transformComponent.m_GlobalRotation.ClearChanged();
        transformComponent.m_GlobalScale.ClearChanged();
        transformComponent.Rotation.ClearChanged();
        transformComponent.Position.ClearChanged();
        transformComponent.Scale.ClearChanged();
    }
}
