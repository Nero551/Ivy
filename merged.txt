struct IArchetype
{
    using ComponentCombo = std::tuple<>;
    virtual ~IArchetype() = default;

    Signature Supported;
    size_t Stride;
    size_t EntityOffset;
    std::unordered_map<Signature, size_t> Offsets;

    virtual void Remove(const Entity& entity) = 0;
    virtual bool Has(const Entity& entity) const = 0;
    virtual void* GetTypeErased(const Entity& entity, Signature typeSig) = 0;
    virtual std::byte* GetBase() = 0;
    virtual unsigned int Size() const = 0;
};

template <typename... Args> requires(U::AreUnique<Args...> && (std::derived_from<Args, Component> && ...))
struct Archetype : IArchetype
{
    using ComponentCombo = std::tuple<Args...>;
    U::SparseSetAoS<ComponentCombo> m_Data{};

    std::tuple<Args...>& Add(const Entity& entity)
    {
        return (*m_Data.Emplace(entity.GetId().Index)).Value;
    }

    Archetype()
    {
        using Entry = typename U::SparseSetAoS<ComponentCombo>::Entry;
        (Supported.set(ComponentId<Args>()), ...);
        Stride = sizeof(Entry);

        Entry entry{};
        EntityOffset = reinterpret_cast<const std::byte*>(&entry.SparseIndex) -
            reinterpret_cast<const std::byte*>(&entry);
        ((Offsets[MakeSignature<Args>()] = GetOffset<Args>(entry)), ...);
    }
    template <typename T> size_t GetOffset(const typename U::SparseSetAoS<ComponentCombo>::Entry& entry)
    {
        const std::byte* base = reinterpret_cast<const std::byte*>(&entry);
        const std::byte* component = reinterpret_cast<const std::byte*>(&std::get<T>(entry.Value));

        return component - base;
    }

    std::byte* GetBase() override
    {
        return reinterpret_cast<std::byte*>(m_Data.DenseData().data());
    }

    unsigned int Size() const override
    {
        return m_Data.Size();
    }

    bool Has(const Entity& entity) const override
    {
        return m_Data.Contains(entity.GetId().Index);
    }

    void Remove(const Entity& entity) override
    {
        m_Data.Erase(entity.GetId().Index);
    }

    void* GetTypeErased(const Entity& entity, Signature typeSig) override
    {
        ComponentCombo& components = Get(entity);
        void* result = nullptr;

        auto GetComponent = [&]<typename T>()
        {
            if (typeSig == MakeSignature<T>())
            {
                result = static_cast<void*>(&std::get<T>(components));
            }
        };

        (GetComponent.template operator()<Args>(), ...);

        return result;
    }
    std::tuple<Args...>& Get(const Entity& entity)
    {
        U::Log::Assert(m_Data.Contains(entity.GetId().Index), "Archetype: [ ", typeid(Args).name()...,
            " ] doesn't contain entity: ", entity.GetId());
        return m_Data[entity.GetId().Index];
    }

    template <typename T> T& GetOne(const Entity& entity) requires(std::is_same_v<T, Args> || ...)
    {
        U::Log::Assert(m_Data.Contains(entity.GetId().Index), "Archetype: [ ", typeid(Args).name()...,
            " ] doesn't contain entity: ", entity.GetId());
        return std::get<T>(m_Data[entity.GetId().Index]);
    }
};
} // namespace Ivy::C

inline unsigned int NextComponentId()
{
    static unsigned int counter = 0;
    U::Log::Assert(counter < 64, "Too many component types for Signature");
    return ++counter;
}

template <typename T> inline unsigned int ComponentId()
{
    static const unsigned int id = NextComponentId();
    return id;
}

using Signature = std::bitset<64>;

template <typename... Args> Signature MakeSignature()
{
    Signature s;
    (s.set(ComponentId<Args>()), ...);
    return s;
}

struct EntityComponentSystem
{
    EntityDestroyed EntityDestroyed{};
    EntityCreated EntityCreated{};

    Query Query{};
    void RemoveEntity(Entity& entity)
    {
        const auto descendants = entity.GetDescendants();

        if (entity.HasParent())
        {
            entity.ClearParent();
        }

        EntityDestroyed.Entity = &entity;
        EntityDestroyed.Fire();
        m_Entities.Erase(entity.GetId().Index);
        Query.RemoveAllComponents(entity);

        for (auto& descendant : descendants)
        {
            EntityDestroyed.Entity = descendant;
            EntityDestroyed.Fire();
            m_Entities.Erase(descendant->GetId().Index);
            Query.RemoveAllComponents(*descendant);
        }
    }

    void ReserveEntities(size_t count)
    {
        m_Entities.ReserveDense(count);
        m_Entities.ReserveSparse(count);
    }

    Entity& GetEntity(EntityID id)
    {
        return m_Entities[id.Index];
    }

    Entity& GetActiveEntity(unsigned int id)
    {
        return m_Entities[id];
    }

    template <typename T> Entity& CreateEntity() requires std::derived_from<T, Entity>
    {
        const EntityID id = m_Ids.Acquire();
        T entity{*this};
        entity.m_Id = id;
        entity.Initialize();

        auto it = m_Entities.Emplace(id.Index, std::move(entity));
        EntityCreated.Entity = &*it;
        EntityCreated.Fire();

        return *it;
    }

    Entity& FindEntity(EntityID id)
    {
        if (!m_Ids.IsAcquired(id))
        {
            U::Log::Fatal("Entity Not Found: ", id);
        }
        return m_Entities[id.Index];
    }

    Entity& FindActiveEntity(unsigned int id)
    {
        if (!m_Entities.Contains(id))
        {
            U::Log::Fatal("Entity Not Found: ", id);
        }
        return m_Entities[id];
    }

    U::CheckedPtr<Entity> TryFindEntity(EntityID id)
    {
        if (!m_Ids.IsAcquired(id))
        {
            return nullptr;
        }
        return &m_Entities[id.Index];
    }

    U::CheckedPtr<Entity> TryFindActiveEntity(unsigned int id)
    {
        if (!m_Entities.Contains(id))
        {
            return nullptr;
        }
        return &m_Entities[id];
    }

    unsigned int EntityAmount() const
    {
        return m_Entities.Size();
    }

  private:
    U::SparseSetSoA<Entity> m_Entities{};
    U::GIndexPool<> m_Ids{};
};

} // namespace Ivy::C

#include "Entity.hpp"

EntityID Entity::GetId() const
{
    return m_Id;
}

Entity& Entity::GetRoot()
{
    U::CheckedPtr<Entity> current = this;

    while (current->HasParent())
    {
        current = &m_ECS.GetEntity(current->m_Parent);
    }

    return *current;
}

void Entity::Destroy()
{
    m_ECS.RemoveEntity(*this);
}

void Entity::AttachChild(Entity& child)
{
    if (HasChild(child))
    {
        U::Log::Fatal("Child Entity already exists: ", child.m_Id.Index);
        return;
    }

    if (&child == this)
    {
        U::Log::Error("An entity cannot be its own child.");
        return;
    }

    if (child.IsDescendantOf(*this))
    {
        U::Log::Error("An entity cannot have its descendant as a child.");
        return;
    }

    if (child.IsAncestorOf(*this))
    {
        U::Log::Error("An entity cannot have its ancestor as a child.");
        return;
    }

    m_Children.Emplace(child.m_Id.Index, child.m_Id);
    child.m_Parent = m_Id;
}

void Entity::DetachChild(Entity& child)
{
    if (HasChild(child))
    {
        m_Children.Erase(child.GetId().Index);
        child.ClearParent();
    }
}

bool Entity::HasChild(const Entity& child) const
{
    return m_Children.Contains(child.GetId().Index) && m_Children[child.GetId().Index] == child.GetId();
}

Entity& Entity::GetChild(const EntityID id) const
{
    return m_ECS.GetEntity(id);
}

Entity& Entity::FindChild(const EntityID id) const
{
    if (m_Children.Contains(id.Index) && m_Children[id.Index] == id)
    {
        return m_ECS.FindEntity(id);
    }

    U::Log::Fatal(std::format("Entity {} has no child {}", m_Id.Index, id.Index));
}

U::CheckedPtr<Entity> Entity::TryFindChild(const EntityID id) const
{
    if (m_Children.Contains(id.Index) && m_Children[id.Index] == id)
    {
        return &m_ECS.GetEntity(id);
    }

    return {std::format("Entity {} has no child {}", m_Id.Index, id.Index)};
}

std::vector<U::CheckedPtr<Entity>> Entity::GetChildren() const
{
    std::vector<U::CheckedPtr<Entity>> children;
    children.reserve(m_Children.Size());

    for (auto id : m_Children)
    {
        children.emplace_back(&m_ECS.GetEntity(id));
    }

    return children;
}

unsigned int Entity::ChildCount() const
{
    return m_Children.Size();
}

void Entity::ForEachChild(std::function_ref<void(Entity&)> callback) const
{
    for (const EntityID id : m_Children)
    {
        callback(m_ECS.GetEntity(id));
    }
}

void Entity::DestroyChild(Entity& child)
{
    if (!HasChild(child))
    {
        return;
    }
    child.Destroy();
}

void Entity::DestroyChildren()
{
    while (!m_Children.Empty())
    {
        DestroyChild(m_ECS.GetEntity(*m_Children.begin()));
    }
}

bool Entity::HasDescendant(const Entity& descendant) const
{
    return IsAncestorOf(descendant);
}

bool Entity::IsDescendantOf(const Entity& ancestor) const
{
    EntityID current = m_Parent;

    while (current != EntityID{0, 0})
    {
        if (current == ancestor.GetId())
        {
            return true;
        }

        current = m_ECS.GetEntity(current).m_Parent;
    }

    return false;
}

std::vector<U::CheckedPtr<Entity>> Entity::GetDescendants() const
{
    std::vector<U::CheckedPtr<Entity>> descendants;
    descendants.reserve(m_Children.Size() * 3);

    RecursiveChildren(descendants, *this);

    return descendants;
}

void Entity::ForEachDescendant(std::function_ref<void(Entity&)> callback)
{
    for (EntityID id : m_Children)
    {
        auto& child = m_ECS.GetEntity(id);
        callback(child);
        child.ForEachDescendant(callback);
    }
}

bool Entity::HasAncestor(const Entity& ancestor) const
{
    return IsDescendantOf(ancestor);
}

bool Entity::IsAncestorOf(const Entity& descendant) const
{
    for (auto id : m_Children)
    {
        if (id == descendant.GetId())
        {
            return true;
        }

        const Entity& child = m_ECS.GetEntity(id);

        if (child.IsAncestorOf(descendant))
        {
            return true;
        }
    }

    return false;
}

std::vector<U::CheckedPtr<Entity>> Entity::GetAncestors() const
{
    std::vector<U::CheckedPtr<Entity>> ancestors;

    EntityID current = m_Parent;

    while (current != EntityID{0, 0})
    {
        Entity& entity = m_ECS.GetEntity(current);
        ancestors.emplace_back(&entity);
        current = entity.m_Parent;
    }

    return ancestors;
}

void Entity::ForEachAncestor(std::function_ref<void(Entity&)> callback)
{
    U::CheckedPtr<Entity> current = TryFindParent();

    while (current)
    {
        callback(*current);
        current = current->TryFindParent();
    }
}

void Entity::SetParent(Entity& parent)
{
    parent.AttachChild(*this);
}

Entity& Entity::GetParent() const
{
    return m_ECS.GetEntity(m_Parent);
}

U::CheckedPtr<Entity> Entity::TryFindParent() const
{
    if (m_Parent == EntityID{0, 0})
    {
        return nullptr;
    }
    return m_ECS.TryFindEntity(m_Parent);
}

bool Entity::HasParent() const
{
    return m_Parent != EntityID{0, 0};
}

void Entity::ClearParent()
{
    if (HasParent())
    {
        GetParent().DetachChild(*this);
    }
}

bool Entity::operator==(Entity& other) const
{
    return GetId() == other.GetId();
}

void Entity::RecursiveChildren(std::vector<U::CheckedPtr<Entity>>& entities, const Entity& entity) const
{
    for (auto id : entity.m_Children)
    {
        Entity& child = m_ECS.GetEntity(id);

        entities.emplace_back(&child);
        RecursiveChildren(entities, child);
    }
}

#pragma once
#include "Archetype.hpp"
#include "Component.hpp"
#include "Entity.hpp"
#include "Utilities/CheckedPtr.hpp"
#include "Utilities/DataStructures/TypedVector.hpp"
#include "Utilities/Debug/Log.hpp"
#include "Utilities/TupleUtils.hpp"
#include <array>
#include <concepts>
#include <cstddef>
#include <memory>
#include <tuple>
#include <typeindex>
#include <utility>
#include <vector>
namespace Ivy::C
{
struct Query
{

    using Cache = std::vector<std::vector<std::pair<unsigned int, std::vector<void*>>>>;
    Query() = default;

    template <typename... Args>
    std::tuple<Args&...> GetComponent(const Entity& entity) const
        requires(U::AreUnique<Args...> && (std::derived_from<Args, Component> && ...) && sizeof...(Args) != 1)
    {
        return std::tie(GetComponent<Args>(entity)...);
    }

    template <typename T>
    T& GetComponent(const Entity& entity) const requires(std::derived_from<T, Component>)
    {
        U::Log::Assert(entity.m_Archetype != nullptr, "Entity is not Connected To Any Archetype");
        if (m_Archetypes.Contains<T>())
        {
            Archetype<T>& archetype = static_cast<Archetype<T>&>(*m_Archetypes.Get<T>());
            if (archetype.Has(entity))
            {
                return archetype.template GetOne<T>(entity);
            }
        }

        const Signature sig = MakeSignature<T>();
        if ((sig & entity.m_Archetype->Supported) == sig)
        {
            return *static_cast<T*>(entity.m_Archetype->GetTypeErased(entity, sig));
        }
        U::Log::Fatal("Component: ", typeid(T).name(), " Doesn't exist for Entity: ", entity.GetId().Index);
    }

    template <typename... Args>
    bool HasComponent(const Entity& entity) const
        requires(U::AreUnique<Args...> && (std::derived_from<Args, Component> && ...))
    {
        if (entity.m_Archetype == nullptr)
        {
            return false;
        }

        const Signature sig = MakeSignature<Args...>();

        return (entity.m_Archetype->Supported & sig).count() == sizeof...(Args);
    }

    void RemoveAllComponents(Entity& entity)
    {
        if (entity.m_Archetype)
        {
            entity.m_Archetype->Remove(entity);
            entity.m_Archetype.Reset();
        }
    }

    template <typename... Args> decltype(auto) SetArchetype(Entity& entity) requires(U::AreUnique<Args...>)
    {
        using Components = U::ConnectArgsIntoTuple<Args...>;

        auto action = [&]<typename... Ts>(Entity& entity) { return SetArchetypeTypes<Ts...>(entity); };

        return U::UnpackAndRun<Components>(
            std::make_index_sequence<std::tuple_size_v<Components>>{}, action, entity);
    }

    template <typename T> T& SetArchetype(Entity& entity) requires(std::derived_from<T, Component>)
    {
        return std::get<T>(SetArchetypeTypes<T>(entity));
    }

    template <typename... Args, typename F>
    requires(U::AreUnique<Args...> && (std::derived_from<Args, Component> && ...)) &&
        std::invocable<F&, unsigned int, Args&...>
    void ForEach(F&& action)
    {
        static const Signature sig = MakeSignature<Args...>();

        for (auto& ptr : m_Archetypes)
        {
            auto& archetype = *ptr;

            if ((sig & archetype.Supported) != sig)
            {
                continue;
            }

            const std::array<size_t, sizeof...(Args)> offsets{archetype.Offsets.at(MakeSignature<Args>())...};

            std::byte* base = archetype.GetBase();

            for (unsigned int i = 0; i < archetype.Size(); ++i)
            {
                std::byte* row = base + i * archetype.Stride;

                unsigned int entityIndex = *reinterpret_cast<unsigned int*>(row + archetype.EntityOffset);

                Invoke<Args...>(action, entityIndex, row, offsets, std::index_sequence_for<Args...>{});
            }
        }
    }

    template <typename... Args, size_t... I>
    static void Invoke(auto& action, unsigned int entityIndex, std::byte* row,
        const std::array<size_t, sizeof...(Args)>& offsets, std::index_sequence<I...>)
    {
        action(entityIndex, *reinterpret_cast<Args*>(row + offsets[I])...);
    }

  private:
    template <typename... Args>
    Archetype<Args...>& GetArchetype() requires(std::derived_from<Args, Component> && ...)
    {
        if (m_Archetypes.Contains<Args...>())
        {
            return static_cast<Archetype<Args...>&>(*m_Archetypes.Get<Args...>());
        };
        return static_cast<Archetype<Args...>&>(
            **m_Archetypes.Emplace<Args...>(std::make_unique<Archetype<Args...>>()));
    }

    template <typename... Args>
    std::tuple<Args&...> SetArchetypeTypes(Entity& entity)
        requires(U::AreUnique<Args...> && (std::derived_from<Args, Component> && ...))
    {
        auto& archetype = GetArchetype<Args...>();

        if (entity.m_Archetype == &archetype)
        {
            return archetype.Get(entity);
        }

        U::CheckedPtr<std::tuple<Args...>> components;

        if (!entity.m_Archetype)
        {
            components = &archetype.Add(entity);
            entity.m_Archetype = &archetype;
        }
        else
        {
            U::CheckedPtr<IArchetype> previous = entity.m_Archetype;

            components = &archetype.Add(entity);

            (
                [&]
                {
                    if ((MakeSignature<Args>() & previous->Supported).count() != 0)
                    {
                        *static_cast<Args*>(archetype.GetTypeErased(entity, MakeSignature<Args>())) =
                            std::move(
                                *static_cast<Args*>(previous->GetTypeErased(entity, MakeSignature<Args>())));
                    }
                }(),
                ...);

            previous->Remove(entity);
            entity.m_Archetype = &archetype;
        }

        //TODO: instead of completely refreshing the cache. i could do a search and \
        // invalidate only caches associated with the modified components. \
        // can make cache a runtime map with type_index as well if i need that.

        return *components;
    }
    U::TypedVector<std::unique_ptr<IArchetype>> m_Archetypes;
};
