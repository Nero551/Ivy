82 {
83 const T m = hyperSpherical.Magnitude;
84
85 Vector result;
86
87 result.x = m * std::cos(hyperSpherical.Elevation) * std::cos(hyperSpherical.Azimuth) *
88 std::cos(hyperSpherical.HyperAngle);
89
90 result.y = m * std::sin(hyperSpherical.Elevation) * std::cos(hyperSpherical.HyperAngle);
91
92 result.z = m * std::cos(hyperSpherical.Elevation) * std::sin(hyperSpherical.Azimuth) *
93 std::cos(hyperSpherical.HyperAngle);

95 result.w = m * std::sin(hyperSpherical.HyperAngle);
96
