235struct OperationDimensional<A, B, Name>
236{
237 using Left = A::Normalized;
238 using Right = B::Normalized;

240 using Flattenend = Concat<typename Left::Flattenend, typename Right::Flattenend>::Type;
241 using Merged = Merge<Flattenend>::Type;
242 using ZeroExpRemoved = RemoveZeroExponents<Merged::Size, Merged>::Type;
243 using Sorted = Sort<ZeroExpRemoved>::Type;
244 using Normalized = Sorted::Rebuild;
245
246 static constexpr int Exponent = 1;

248 template <int E>
249 using WithExponent = OperationDimensional<Exponentiate<Left, Left::Exponent * E>,
250 Exponentiate<Right, Right::Exponent * E>>;

253 using WithRoot =
254 OperationDimensional<typename Left::template WithRoot<E>, typename Right::template WithRoot<E>>;

256 static std::ostream& Print(std::ostream& os)
257 {
258 if constexpr (!Name.Empty())
259 {
260 return os << std::string_view(Name);
261 }
262 else if constexpr (!IsOperation<Normalized>)
263 {
264 return Normalized::Print(os);
265 }
266
267 using L = Normalized::Left;
268 using R = Normalized::Right;
269
270 if constexpr (L::Exponent < 0 && R::Exponent < 0)
271 {
272 os << "1/(";
273
274 Exponentiate<L, -L::Exponent>::Print(os);
275 Exponentiate<R, -R::Exponent>::Print(os);
276
277 return os << ")";
278 }

280 {
281 R::Print(os);
