54 static constexpr Quaternion FromQPolar(const QPolar<T>& qPolar)
55 {
56 Quaternion result;

60 result.x = m * (qPolar.Axis.x * sine);

77 static constexpr Quaternion FromMatrix3(const Matrix<3, 3, T>& mat3)
78 {
79 const T trace = mat3(0, 0) + mat3(1, 1) + mat3(2, 2);
80
81 Quaternion rotation;
82
83 if (trace > 0)
84 {
85 const T s = Sqrt(trace + T{1}) * T{2};
86
87 rotation.w = T{0.25} * s;
88 rotation.x = (mat3(2, 1) - mat3(1, 2)) / s;
89 rotation.y = (mat3(0, 2) - mat3(2, 0)) / s;
90 rotation.z = (mat3(1, 0) - mat3(0, 1)) / s;
91 }
92 else if (mat3(0, 0) > mat3(1, 1) && mat3(0, 0) > mat3(2, 2))
93 {
94 const T s = Sqrt(T{1} + mat3(0, 0) - mat3(1, 1) - mat3(2, 2)) * T{2};
95
96 rotation.w = (mat3(2, 1) - mat3(1, 2)) / s;
97 rotation.x = T{0.25} * s;
98 rotation.y = (mat3(0, 1) + mat3(1, 0)) / s;
99 rotation.z = (mat3(0, 2) + mat3(2, 0)) / s;
100 }
101 else if (mat3(1, 1) > mat3(2, 2))
102 {
103 const T s = Sqrt(T{1} + mat3(1, 1) - mat3(0, 0) - mat3(2, 2)) * T{2};
104
105 rotation.w = (mat3(0, 2) - mat3(2, 0)) / s;
106 rotation.x = (mat3(0, 1) + mat3(1, 0)) / s;
107 rotation.y = T{0.25} * s;
108 rotation.z = (mat3(1, 2) + mat3(2, 1)) / s;
109 }
110 else
111 {
112 const T s = Sqrt(T{1} + mat3(2, 2) - mat3(0, 0) - mat3(1, 1)) * T{2};
113
114 rotation.w = (mat3(1, 0) - mat3(0, 1)) / s;
115 rotation.x = (mat3(0, 2) + mat3(2, 0)) / s;
116 rotation.y = (mat3(1, 2) + mat3(2, 1)) / s;
117 rotation.z = T{0.25} * s;
118 }
119
120 // Convert the rotation quaternion from half-angle form
121 // to the full-angle quaternion representation.
122 return rotation * rotation;
123 }

125 static Quaternion FromTo(const Vector<3, T>& from, const Vector<3, T>& to)
126 {
127 Vector<3, T> A = from.Normalized();
128 Vector<3, T> B = to.Normalized();
129
130 T Dot = A.Dot(B);
131
132 if (Dot >= T(1.0) - EPSILON)
133 {
134 return Identity();
135 }
136
137 if (Dot <= T(-1.0) + EPSILON)
138 {
139 Vector<3, T> Axis = A.Cross(Vector<3, T>::Right()).Normalized();
140
141 if (Axis.LengthSquared() <= EPSILON)
142 {
143 Axis = A.Cross(Vector<3, T>::Up()).Normalized();
144 }
145
146 return FromQPolar({Axis, M::PI});
147 }
148
149 Vector<3, T> Axis = A.Cross(B).Normalized();
150 T Angle = std::acos(std::clamp(Dot, T(-1.0), T(1.0)));
151
152 return FromQPolar({Axis, Angle});
153 }

296 constexpr Matrix<4, 4, T> ToMatrix4() const
297 {
298 const auto q = Normalized();
299 const T wHalf = Sqrt((T{1} + q.w) * T{0.5});
300
301 if (wHalf < EPSILON)
302 {
303 return Matrix<4, 4, T>::Identity().RotateAroundAxis(Axis(), Angle());
304 }
305
306 const T invTwoW = T{1} / (T{2} * wHalf);
307 const T hx = q.x * invTwoW;
308 const T hy = q.y * invTwoW;
309 const T hz = q.z * invTwoW;
310
311 const T xx = hx * hx, yy = hy * hy, zz = hz * hz;
312 const T xy = hx * hy, xz = hx * hz, yz = hy * hz;
313 const T wx = wHalf * hx, wy = wHalf * hy, wz = wHalf * hz;
314
315 Matrix<4, 4, T> m = Matrix<4, 4, T>::Identity();
316 m(0, 0) = T{1} - T{2} * (yy + zz);
317 m(0, 1) = T{2} * (xy - wz);
318 m(0, 2) = T{2} * (xz + wy);
319 m(1, 0) = T{2} * (xy + wz);
320 m(1, 1) = T{1} - T{2} * (xx + zz);
321 m(1, 2) = T{2} * (yz - wx);
322 m(2, 0) = T{2} * (xz - wy);
323 m(2, 1) = T{2} * (yz + wx);
324 m(2, 2) = T{1} - T{2} * (xx + yy);
325 return m;
326 }

598template <Scalar T> constexpr Quaternion<T> Exp(const Quaternion<T>& q)
599{
600 T r = q.Magnitude();
601 T theta = q.Angle();
602 Vector<3, T> axis = q.Axis();
603
604 T magnitude = Exp(r * std::cos(theta));
605 T rsin = r * std::sin(theta);
606
607 Quaternion<T> result;
608 result.w = magnitude * std::cos(rsin);
609 result.x = magnitude * axis.x * std::sin(rsin);
610 result.y = magnitude * axis.y * std::sin(rsin);
611 result.z = magnitude * axis.z * std::sin(rsin);
612
613 return result;
614}

651template <Scalar T> constexpr Quaternion<T> Ln(const Quaternion<T>& q)
652{
653 Quaternion<T> result;
654 Vector<3, T> axis = q.Axis();
655
656 // q.w is m * cos(rsin(x))
657 // their arctan returns rsin, which is the original magnitude * sin(original angle)
658 T rsin = std::atan2(q.Magnitude() * std::sin(q.Angle()), q.w);
659
660 // axis is unchanged by exponentiation so "u" remains the same
661 // so this is u * original magnitude * sin(original angle)
662 result.w = Ln(q.Magnitude());
663 result.x = axis.x * rsin;
664 result.y = axis.y * rsin;
665 result.z = axis.z * rsin;
666
667 return result;
