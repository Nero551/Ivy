97template <int Variables, int Equations = Variables, Scalar T = float> requires(Equations > 1 && Variables > 0)
98struct LinearSystem
99{
101 template <typename... Args>
102 constexpr LinearSystem(Args... equations)
103 requires(sizeof...(Args) == Equations && (std::same_as<Args, LinearEquation<Variables, T>> && ...))
104 : m_Equations{equations...}
105 {
106 }

109 constexpr LinearSolution<Variables, T> Solve() const
110 {
111 Matrix<Equations, Variables, T> coefficientMatrix;
112 Matrix<Equations, Variables + 1, T> augmentedMatrix;
113
114 //construct the coefficient matrix and the augmented row echelon form matrix.
115 for (int r = 0; r < Equations; ++r)
116 {
117 auto& equation = (*this)(r);
118 for (int c = 0; c < Variables; ++c)
119 {
120 augmentedMatrix(r, c) = equation(c);
121 coefficientMatrix(r, c) = equation(c);
122 }
123 augmentedMatrix(r, Variables) = equation.Constant;
124 }
125 augmentedMatrix = augmentedMatrix.RowEchelon();
126
127 switch (DetermineSolutionType(coefficientMatrix, augmentedMatrix))
128 {
129 case SolutionType::None:
130 return LinearSolution<Variables, T>{SolutionType::None, Set<Vector<Variables, T>>::Empty()};
131
132 case SolutionType::Unique:
133 return LinearSolution<Variables, T>{SolutionType::Unique, BackSubstitute(augmentedMatrix)};
134
135 case SolutionType::Infinite:
136 return LinearSolution<Variables, T>{SolutionType::Infinite, CreateSolutionSet()};
137
138 default:
139 U::Log::Error("LinearSystem: Unknown solution type.");
140 }
141 }

165 static constexpr SolutionType DetermineSolutionType(
166 const Matrix<Equations, Variables, T>& coefficientMatrix,
167 const Matrix<Equations, Variables + 1, T>& augmentedMatrix)
168 {
169 const unsigned int coefficientRank = coefficientMatrix.Rank();
170 const unsigned int augmentedRank = augmentedMatrix.Rank();
171
172 if (coefficientRank != augmentedRank)
173 {
174 return SolutionType::None;
175 }
176
177 if (coefficientRank < Variables)
178 {
179 return SolutionType::Infinite;
180 }
181
182 return SolutionType::Unique;
183 }

186 constexpr Set<Vector<Variables, T>> CreateSolutionSet() const
187 {
188 auto equations = Data();
189 return Set<Vector<Variables, T>>{[equations](const Vector<Variables, T>& x)
190 {
191 for (const auto& equation : equations)
192 {
193 T result = -equation.Constant;
194
195 for (int variable = 0; variable < Variables; ++variable)
196 {
197 result += equation(variable) * x(variable);
198 }
199
200 if (!NearlyEquals(result, T{0}))
201 {
202 return false;
203 }
204 }
205
206 return true;
207 }};
208 }
209
211 constexpr Vector<Variables, T> BackSubstitute(
212 const Matrix<Equations, Variables + 1, T>& augmentedMatrix) const
213 {
214 Vector<Variables, T> solution;
215 for (int row = Variables - 1; row >= 0; --row)
216 {
217 T value = augmentedMatrix(row, Variables);
218
219 for (int column = row + 1; column < Variables; ++column)
220 {
221 value -= augmentedMatrix(row, column) * solution(column);
222 }
223
224 solution(row) = value / augmentedMatrix(row, row);
225 }
226 return solution;
227 }
