10template <typename T> consteval std::string_view GetTypeName()
11{
12 constexpr std::string_view Function = __PRETTY_FUNCTION__;
13 constexpr std::string_view Prefix = "T = ";
14 constexpr std::size_t Start = Function.find(Prefix) + Prefix.size();
15 constexpr std::size_t End = Function.find(']', Start);
16
17 return Function.substr(Start, End - Start);
18}

20consteval unsigned long Hash(const std::string_view typeName)
21{
22 unsigned long hashValue = 0xcbf29ce484222325ULL;
23 for (char c : typeName)
24 {
25 constexpr unsigned long prime = 0x100000001b3ULL;
26 hashValue ^= static_cast<unsigned long>(c);
27 hashValue *= prime;
28 }
29 return hashValue;
30}
