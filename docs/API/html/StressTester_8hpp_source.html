51 void TwoDimensionalProjection(int increase)
52 {
53 points.reserve(M::Pow(360.0f / increase, 1));
54 U::Log::Info(M::Pow(360.0f / increase, 1));
55 for (int theta = -180; theta < 180; theta += increase)
56 {
57 M::Vector<2> v2 = M::Vector<2>::FromPolar(M::Polar(M::Rad(theta)));
58 float proj = v2.StereoProject();
59 auto& point = Plot({proj, 0, 0});
60 points.emplace_back(point.GetId());
61 }
62 }
63 void ThreeDimensionalProjection(int increase)
64 {
65 U::Log::Info(M::Pow(360.0f / increase, 2));
66 points.reserve(M::Pow(360.0f / increase, 2));
67
68 for (int theta = -180; theta < 180; theta += increase)
69 {
70 for (int phi = -180; phi < 180; phi += increase)
71 {
72 M::Vector<3> v3 = M::Vector<3>::FromSpherical(M::Spherical(M::Rad(theta), M::Rad(phi)));
73 M::Vector<2> proj = v3.StereoProject();
74 auto& point = Plot({proj.x, proj.y, 0});
75 points.emplace_back(point.GetId());
76 }
77 }
78 }
79 void FourDimensionalProjection(int increase)
80 {
81 points.reserve(M::Pow(360.0f / increase, 3));
82 U::Log::Info(M::Pow(360.0f / increase, 3));
83 C::World::Get().ReserveEntities(M::Pow(360.0f / increase, 3));
84 for (int theta = -180; theta < 180; theta += increase)
85 {
86 for (int phi = -180; phi < 180; phi += increase)
87 {
88 for (int h = -180; h < 180; h += increase)
89 {
90 M::Vector<4> v4 = M::Vector<4>::FromHyperSpherical(
91 M::HyperSpherical(M::Rad(theta), M::Rad(phi), M::Rad(h)));
92 M::Vector<3> proj = v4.StereoProject();
93 auto& point = Plot(proj);
94 points.emplace_back(point.GetId());
95 }
96 }
