109 constexpr Matrix RotateAroundAxis(const Vector<3, T>& axis, const T radian) const
110 {
111 const Vector<3, T> forward = axis.Normalized();
112
113 const T cosine = std::cos(radian);
114 const T sine = std::sin(radian);
115 const T oneMinusCosine = 1.0f - cosine;
116
117 const T x = forward(0);
118 const T y = forward(1);
119 const T z = forward(2);
120
121 Matrix rotationMatrix = Matrix::Identity();
122
123 rotationMatrix(0, 0) = oneMinusCosine * x * x + cosine;
124 rotationMatrix(0, 1) = oneMinusCosine * x * y - sine * z;
125 rotationMatrix(0, 2) = oneMinusCosine * x * z + sine * y;
126
127 rotationMatrix(1, 0) = oneMinusCosine * x * y + sine * z;
128 rotationMatrix(1, 1) = oneMinusCosine * y * y + cosine;
129 rotationMatrix(1, 2) = oneMinusCosine * y * z - sine * x;
130
131 rotationMatrix(2, 0) = oneMinusCosine * x * z - sine * y;
132 rotationMatrix(2, 1) = oneMinusCosine * y * z + sine * x;
133 rotationMatrix(2, 2) = oneMinusCosine * z * z + cosine;
134
135 return *this * rotationMatrix;
136 }

163 static constexpr Matrix Perspective(const T fovRad, const T aspectRatio, const T near, const T far)
164 {
165 Matrix matrix = Matrix::Zero();
166
167 const T f = 1.0f / std::tan(fovRad * 0.5f);
168
169 matrix(0, 0) = f / aspectRatio;
170 matrix(1, 1) = f;
171
172 matrix(2, 2) = -(far + near) / (far - near);
173 matrix(2, 3) = -(2.0f * far * near) / (far - near);
174
175 matrix(3, 2) = -1.0f;
176

181 static constexpr Matrix LookAt(
182 const Vector<3, T>& position, const Vector<3, T>& target, const Vector<3, T>& up)
183 {
184 const Vector<3, T> forward = (target - position).Normalized();
185 const Vector<3, T> right = forward.Cross(up).Normalized();
186 const Vector<3, T> correctedUp = right.Cross(forward);
187
188 Matrix basisMatrix = Matrix::Identity();
189
190 basisMatrix(0, 0) = right.x;
191 basisMatrix(1, 0) = right.y;
192 basisMatrix(2, 0) = right.z;
193
194 basisMatrix(0, 1) = correctedUp.x;
195 basisMatrix(1, 1) = correctedUp.y;
196 basisMatrix(2, 1) = correctedUp.z;
197
198 basisMatrix(0, 2) = -forward.x;
199 basisMatrix(1, 2) = -forward.y;
200 basisMatrix(2, 2) = -forward.z;
201
202 Matrix result = basisMatrix.Transpose();
203 result(0, 3) = -position.Dot(right);
204 result(1, 3) = -position.Dot(correctedUp);
205 result(2, 3) = position.Dot(forward);
206 return result;
207 }

217 constexpr Matrix Inverse() const
218 {
219 const T m0 = (*this)(0, 0);
220 const T m1 = (*this)(0, 1);
221 const T m2 = (*this)(0, 2);
222 const T m3 = (*this)(0, 3);
223 const T m4 = (*this)(1, 0);
224 const T m5 = (*this)(1, 1);
225 const T m6 = (*this)(1, 2);
226 const T m7 = (*this)(1, 3);
227 const T m8 = (*this)(2, 0);
228 const T m9 = (*this)(2, 1);
229 const T m10 = (*this)(2, 2);
230 const T m11 = (*this)(2, 3);
231 const T m12 = (*this)(3, 0);
232 const T m13 = (*this)(3, 1);
233 const T m14 = (*this)(3, 2);
234 const T m15 = (*this)(3, 3);
235
236 const T s0 = m10 * m15 - m11 * m14;
237 const T s1 = m9 * m15 - m11 * m13;
238 const T s2 = m9 * m14 - m10 * m13;
239 const T s3 = m8 * m15 - m11 * m12;
240 const T s4 = m8 * m14 - m10 * m12;
241 const T s5 = m8 * m13 - m9 * m12;
242
243 Matrix adjugate;
244
245 adjugate(0, 0) = m5 * s0 - m6 * s1 + m7 * s2;
246 adjugate(0, 1) = -(m1 * s0 - m2 * s1 + m3 * s2);
247 adjugate(0, 2) = m13 * (m2 * m7 - m3 * m6) - m14 * (m1 * m7 - m3 * m5) + m15 * (m1 * m6 - m2 * m5);
248 adjugate(0, 3) = -(m9 * (m2 * m7 - m3 * m6) - m10 * (m1 * m7 - m3 * m5) + m11 * (m1 * m6 - m2 * m5));
249
250 adjugate(1, 0) = -(m4 * s0 - m6 * s3 + m7 * s4);
251 adjugate(1, 1) = m0 * s0 - m2 * s3 + m3 * s4;
252 adjugate(1, 2) = -(m12 * (m2 * m7 - m3 * m6) - m14 * (m0 * m7 - m3 * m4) + m15 * (m0 * m6 - m2 * m4));
253 adjugate(1, 3) = m8 * (m2 * m7 - m3 * m6) - m10 * (m0 * m7 - m3 * m4) + m11 * (m0 * m6 - m2 * m4);
254
255 adjugate(2, 0) = m4 * s1 - m5 * s3 + m7 * s5;
256 adjugate(2, 1) = -(m0 * s1 - m1 * s3 + m3 * s5);
257 adjugate(2, 2) = m12 * (m1 * m7 - m3 * m5) - m13 * (m0 * m7 - m3 * m4) + m15 * (m0 * m5 - m1 * m4);
258 adjugate(2, 3) = -(m8 * (m1 * m7 - m3 * m5) - m9 * (m0 * m7 - m3 * m4) + m11 * (m0 * m5 - m1 * m4));
259
260 adjugate(3, 0) = -(m4 * s2 - m5 * s4 + m6 * s5);
261 adjugate(3, 1) = m0 * s2 - m1 * s4 + m2 * s5;
262 adjugate(3, 2) = -(m12 * (m1 * m6 - m2 * m5) - m13 * (m0 * m6 - m2 * m4) + m14 * (m0 * m5 - m1 * m4));
263 adjugate(3, 3) = m8 * (m1 * m6 - m2 * m5) - m9 * (m0 * m6 - m2 * m4) + m10 * (m0 * m5 - m1 * m4);
264
265 const T determinant =
266 m0 * adjugate(0, 0) + m1 * adjugate(1, 0) + m2 * adjugate(2, 0) + m3 * adjugate(3, 0);
267
268 if (std::abs(determinant) < EPSILON)
269 {
270 U::Log::Error("Matrix is not invertible");
271 }
272
273 return adjugate / determinant;
274 }

278 {
279 Matrix<3, 3, T> result;

281
